/**
 * 判斷是否為音訊或影音媒體檔案 (擴大支援 YouTube 下載、MP4、WebM、無副檔名二進位等)
 */
function getSafeFileSize(file) {
  try {
    var mime = (file.getMimeType() || "").toLowerCase();
    if (mime.indexOf('google-apps') !== -1) return 0;
    return file.getSize();
  } catch(e) {
    return 0;
  }
}

/**
 * 判斷是否為音訊或影音媒體檔案 (排除試算表，擴大支援 YouTube、MP4、二進位等)
 */
function isMediaFile(file) {
  try {
    var name = file.getName().toLowerCase();
    var mime = (file.getMimeType() || "").toLowerCase();

    // 排除播放清單設定檔與系統設定
    if (name === 'playlists.json' || name.endsWith('.json')) return false;

    // 絕對排除 Google 原生試算表/簡報/表單
    if (mime.indexOf('google-apps') !== -1) return false;

    // 1. 音訊格式
    if (mime.indexOf('audio') !== -1) return true;
    if (name.endsWith('.mp3') || name.endsWith('.m4a') || name.endsWith('.wav') ||
        name.endsWith('.aac') || name.endsWith('.ogg') || name.endsWith('.opus') ||
        name.endsWith('.flac') || name.endsWith('.wma')) return true;

    // 2. 影音 / YouTube 格式 (MP4, WebM, MKV 等均可在瀏覽器音訊播放器串流)
    if (mime.indexOf('video') !== -1) return true;
    if (name.endsWith('.mp4') || name.endsWith('.webm') || name.endsWith('.mkv') ||
        name.endsWith('.mov') || name.endsWith('.m4v') || name.endsWith('.avi')) return true;

    var size = getSafeFileSize(file);

    // 3. YouTube 下載無副檔名或串流二進位
    if (mime === 'application/octet-stream' && size > 100 * 1024) return true;

    // 4. 大於 1MB 且非文件/試算表之檔案，一律視為影音媒體
    if (size > 1024 * 1024) {
      if (mime !== MimeType.PLAIN_TEXT &&
          mime !== MimeType.GOOGLE_DOCS &&
          mime !== MimeType.GOOGLE_SHEETS &&
          mime !== 'application/pdf' &&
          !name.endsWith('.txt') && !name.endsWith('.md')) {
        return true;
      }
    }
  } catch(e) {
    console.log("isMediaFile error: " + e);
  }
  return false;
}

/**
 * 判斷是否為文章教材檔案
 */
function isArticleFile(file) {
  try {
    var name = file.getName().toLowerCase();
    var mime = (file.getMimeType() || "").toLowerCase();

    // 排除播放清單設定檔與系統設定
    if (name === 'playlists.json' || name.endsWith('.json')) return false;

    if (mime === MimeType.GOOGLE_SHEETS || mime.indexOf('spreadsheet') !== -1) return false;
    if (mime === MimeType.PLAIN_TEXT || name.endsWith('.txt') || name.endsWith('.md')) return true;
    if (mime === MimeType.GOOGLE_DOCS) return true;
    if (mime === 'application/pdf' || name.endsWith('.pdf')) return true;
  } catch(e) {}
  return false;
}

/**
 * 輔助函式：將單一檔案安全加入資產清單 (自動去重，並對 KLine 影音建立雙軌支援)
 */
function addFileToAssets(file, folderTag, assets) {
  try {
    var fileId = file.getId();

    var alreadyInAudios = false;
    for (var i = 0; i < assets.audios.length; i++) {
      if (assets.audios[i].id === fileId) {
        alreadyInAudios = true;
        break;
      }
    }

    var alreadyInArticles = false;
    for (var j = 0; j < assets.articles.length; j++) {
      if (assets.articles[j].id === fileId || assets.articles[j].id === (fileId + "_lesson")) {
        alreadyInArticles = true;
        break;
      }
    }

    var fileName = file.getName();
    var mime = (file.getMimeType() || "").toLowerCase();
    var size = getSafeFileSize(file);
    var fullPath = folderTag + " / " + fileName;
    var streamUrl = "https://drive.google.com/uc?export=download&id=" + fileId + "&confirm=t";

    // 1. 若為媒體檔案 (音樂 / 影片 / YouTube 檔)
    if (isMediaFile(file) && !alreadyInAudios) {
      assets.audios.push({
        id: fileId,
        name: fileName,
        fullPath: fullPath,
        size: size,
        folder: folderTag,
        streamUrl: streamUrl,
        isLarge: (size > 25 * 1024 * 1024)
      });
    }

    // 2. 若為文字/教材檔案
    if (isArticleFile(file) && !alreadyInArticles) {
      var content = "";
      try {
        if (mime === MimeType.GOOGLE_DOCS) {
          content = DocumentApp.openById(fileId).getBody().getText();
        } else if (mime === MimeType.PLAIN_TEXT || fileName.endsWith('.txt') || fileName.endsWith('.md')) {
          content = file.getBlob().getDataAsString("UTF-8");
        } else {
          content = "📄 檔案名稱：" + fileName + "\n檔案大小：" + (size / 1024).toFixed(1) + " KB\n(特殊格式教材)";
        }
      } catch(readErr) {
        content = "（教材讀取中...）";
      }

      assets.articles.push({
        id: fileId,
        name: fileName.replace(/\.[^/.]+$/, ""),
        fullName: fullPath,
        content: content,
        folder: folderTag
      });
    }
    // 3. 針對 KLine 資料夾中的大影音檔，同步在 VoiceTube 建立精讀雙語課程！
    else if ((folderTag === "KLine" || folderTag.indexOf("KLine") !== -1) && !alreadyInArticles) {
      var cleanTitle = fileName.replace(/\.[^/.]+$/, "");
      var klineStudyText = cleanTitle + "。\n\n" +
        "1. Wyckoff Method and Price Action Technical Analysis.\n" +
        "2. Moving averages often lag behind real-time institutional order flow.\n" +
        "3. Three physical footprints reveal the truth behind account drawdown and compounding returns.\n" +
        "4. Professional traders focus on volume price spread analysis rather than lagging indicators.\n" +
        "5. Capital preservation and disciplined risk reward ratios build long-term consistency.";

      assets.articles.push({
        id: fileId + "_lesson",
        name: cleanTitle,
        fullName: fullPath,
        content: klineStudyText,
        folder: "KLine",
        audioId: fileId
      });
    }
  } catch(e) {
    console.log("addFileToAssets 異常: " + e);
  }
}

/**
 * 遞迴遍歷單一資料夾
 */
function scanFolder(folder, folderTag, assets) {
  try {
    var files = folder.getFiles();
    while (files.hasNext()) {
      try {
        var file = files.next();
        addFileToAssets(file, folderTag, assets);
      } catch(fileErr) {
        console.log("略過檔案: " + fileErr);
      }
    }

    var subFolders = folder.getFolders();
    while (subFolders.hasNext()) {
      try {
        var sub = subFolders.next();
        var subName = sub.getName();
        var nextTag = folderTag ? (folderTag + " / " + subName) : subName;
        scanFolder(sub, nextTag, assets);
      } catch(subErr) {
        console.log("略過子目錄: " + subErr);
      }
    }
  } catch(e) {
    console.log("scanFolder 異常: " + e);
  }
}

/**
 * 核心資產掃描 API：直達 KLine、Englishvoice 與主目錄 (精準 100% 辨識)
 */
function getFolderAssets(folderId, path) {
  var assets = { audios: [], articles: [], folders: ["Englishvoice", "KLine"] };

  // 1. 直達掃描 KLine 專屬資料夾 (ID: 1Ey3cePs8PQ5_pUAkqcNDg-2u4BbG5sMH)
  try {
    var klineFolder = DriveApp.getFolderById(KLINE_FOLDER_ID);
    scanFolder(klineFolder, "KLine", assets);
  } catch(e) {
    console.log("讀取 KLine 資料夾異常: " + e);
  }

  // 2. 直達掃描 Englishvoice 資料夾 (ID: 132cXwwdjBGGYyxrYN1V1QMz8SrCNwNwL)
  try {
    var evFolder = DriveApp.getFolderById(ENGLISHVOICE_FOLDER_ID);
    scanFolder(evFolder, "Englishvoice", assets);
  } catch(e) {
    console.log("讀取 Englishvoice 資料夾異常: " + e);
  }

  // 3. 掃描父目錄 (English_voice: 1ra0hX3MuMOm_FYPlj6GpDD4j7t_Gre7)
  try {
    var parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    var parentFiles = parentFolder.getFiles();
    while (parentFiles.hasNext()) {
      try {
        var pf = parentFiles.next();
        addFileToAssets(pf, "English_voice", assets);
      } catch(pfErr) {
        console.log("略過父目錄檔案: " + pfErr);
      }
    }

    var parentSubs = parentFolder.getFolders();
    while (parentSubs.hasNext()) {
      try {
        var sub = parentSubs.next();
        var subId = sub.getId();
        var subName = sub.getName();
        if (subId !== KLINE_FOLDER_ID && subId !== ENGLISHVOICE_FOLDER_ID) {
          if (assets.folders.indexOf(subName) === -1) {
            assets.folders.push(subName);
          }
          scanFolder(sub, subName, assets);
        }
      } catch(subErr) {
        console.log("略過父目錄子資料夾: " + subErr);
      }
    }
  } catch(e) {
    console.log("讀取父資料夾異常: " + e);
  }

  return assets;
}

/**
 * 將音檔轉為 Base64 Data URL (大檔案大於 25MB 自動啟用 Direct Stream URL 突破記憶體限制)
 */
function getAudioBase64(fileId) {
  try {
    var file = DriveApp.getFileById(fileId);
    var size = getSafeFileSize(file);

    if (size > 25 * 1024 * 1024) {
      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      } catch(shareErr) {}
      return "URL:https://drive.google.com/uc?export=download&id=" + fileId + "&confirm=t";
    }

    var mimeType = file.getMimeType();
    var blob = file.getBlob();
    var base64Data = Utilities.base64Encode(blob.getBytes());
    return "data:" + mimeType + ";base64," + base64Data;
  } catch (e) {
    try {
      var f = DriveApp.getFileById(fileId);
      f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch(err2) {}
    return "URL:https://drive.google.com/uc?export=download&id=" + fileId + "&confirm=t";
  }
}

function processArticle(rawText) {
  if (!rawText) return { sentences: [], translations: [] };

  var cleanText = rawText.replace(/\n+/g, ' ');
  var matchArray = cleanText.match(/[^.!?]+[.!?]+/g);
  var sentences = matchArray ? matchArray.map(function(s) { return s.trim(); }) : [cleanText.trim()];
  sentences = sentences.filter(function(s) { return s.length > 0; });

  if (sentences.length === 0) return { sentences: [], translations: [] };

  try {
    var separator = " ||| ";
    var translatedText = LanguageApp.translate(sentences.join(separator), 'en', 'zh-TW');
    var translations = translatedText.split("|||").map(function(s) { return s.trim(); });
    return { sentences: sentences, translations: translations };
  } catch (err) {
    return { sentences: sentences, translations: sentences.map(function() { return "(翻譯中...)"; }) };
  }
}

/**
 * 將文章儲存至 Google Drive 專屬資料夾 (精準支援 KLine 與 Englishvoice 目錄)
 */
function saveArticleToDrive(filename, content, targetFolderName) {
  try {
    if (!filename || !content) return "請輸入檔名與內容";
    var rootFolder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    var targetFolder = rootFolder;
    var destName = targetFolderName || "Englishvoice";

    if (destName === "KLine") {
      try {
        targetFolder = DriveApp.getFolderById(KLINE_FOLDER_ID);
      } catch(e) {
        var subs = rootFolder.getFoldersByName("KLine");
        targetFolder = subs.hasNext() ? subs.next() : rootFolder.createFolder("KLine");
      }
    } else {
      try {
        targetFolder = DriveApp.getFolderById(ENGLISHVOICE_FOLDER_ID);
      } catch(e) {
        var subs = rootFolder.getFoldersByName("Englishvoice");
        targetFolder = subs.hasNext() ? subs.next() : rootFolder.createFolder("Englishvoice");
      }
    }

    var finalName = filename.toLowerCase().endsWith('.txt') ? filename : filename + ".txt";

    var files = targetFolder.getFilesByName(finalName);
    if (files.hasNext()) {
      var file = files.next();
      file.setContent(content);
      return "已成功更新檔案至「" + destName + "」資料夾：" + finalName;
    } else {
      targetFolder.createFile(finalName, content, MimeType.PLAIN_TEXT);
      return "已成功儲存新檔案至「" + destName + "」資料夾：" + finalName;
    }
  } catch (e) {
    return "儲存失敗：" + e.toString();
  }
}

}

/**
 * 儲存生字卡
 */
function apiSaveVocab(vocabItem) {
  try {
    var props = PropertiesService.getUserProperties();
    var raw = props.getProperty('OSCARMARK_VOCAB_VAULT');
    var list = raw ? JSON.parse(raw) : [];

    var foundIndex = -1;
    for (var i = 0; i < list.length; i++) {
      if (list[i].word.toLowerCase() === vocabItem.word.toLowerCase()) {
        foundIndex = i;
        break;
      }
    }

    vocabItem.updated_at = Utilities.formatDate(new Date(), "GMT+8", "yyyy/MM/dd HH:mm");
    if (foundIndex >= 0) {
      list[foundIndex] = Object.assign(list[foundIndex], vocabItem);
    } else {
      vocabItem.id = new Date().getTime();
      vocabItem.mastered = false;
      vocabItem.created_at = vocabItem.updated_at;
      list.unshift(vocabItem);
    }

    props.setProperty('OSCARMARK_VOCAB_VAULT', JSON.stringify(list));
    return { success: true, count: list.length, item: vocabItem };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

    return { success: false, error: err.toString() };
  }
}

/**
 * ==================== GOOGLE DRIVE 雲端播放清單跨裝置同步 ====================
 * 所有播放清單均永久儲存在 Google Drive 根目錄的 playlists.json 中
 * 換筆電、換電腦或不同瀏覽器登入時，自動同步所有自訂歌單！
 */
function apiGetPlaylists() {
  try {
    var folder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    var files = folder.getFilesByName("playlists.json");
    if (files.hasNext()) {
      var file = files.next();
      var content = file.getBlob().getDataAsString("UTF-8");
      if (content && content.trim()) {
        var parsed = JSON.parse(content);
        return { success: true, data: parsed, source: "drive_file" };
      }
    }
    // 備援：若 Drive 檔案尚未建立，檢查 UserProperties
    var props = PropertiesService.getUserProperties();
    var raw = props.getProperty('OSCARMARK_USER_PLAYLISTS');
    if (raw) {
      return { success: true, data: JSON.parse(raw), source: "user_properties" };
    }
    return { success: true, data: null, source: "none" };
  } catch (err) {
    console.log("apiGetPlaylists 錯誤: " + err);
    return { success: false, error: err.toString(), data: null };
  }
}

function apiSavePlaylists(playlists) {
  try {
    if (!playlists || !Array.isArray(playlists)) {
      return { success: false, error: "無效的歌單資料" };
    }
    var folder = DriveApp.getFolderById(PARENT_FOLDER_ID);
    var jsonStr = JSON.stringify(playlists, null, 2);
    var files = folder.getFilesByName("playlists.json");
    if (files.hasNext()) {
      var file = files.next();
      file.setContent(jsonStr);
    } else {
      folder.createFile("playlists.json", jsonStr, MimeType.PLAIN_TEXT);
    }
    // 同時雙重快取備份至 UserProperties
    var props = PropertiesService.getUserProperties();
    props.setProperty('OSCARMARK_USER_PLAYLISTS', jsonStr);
    return { success: true, count: playlists.length };
  } catch (err) {
    console.log("apiSavePlaylists 錯誤: " + err);
    return { success: false, error: err.toString() };
  }
}
